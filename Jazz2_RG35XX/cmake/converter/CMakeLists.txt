cmake_minimum_required(VERSION 3.15)
project(RG35XXAssetPacker C CXX)
if(NOT ENGINE_DIR OR NOT ZLIB_SOURCE_DIR)
    message(FATAL_ERROR "ENGINE_DIR and ZLIB_SOURCE_DIR are required")
endif()
set(NCINE_SOURCE_DIR "${ENGINE_DIR}/Sources")
set(NCINE_VERSION "3.7.0")
find_package(Threads REQUIRED)
set(SKIP_INSTALL_ALL ON CACHE BOOL "" FORCE)
add_subdirectory("${ZLIB_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/zlib" EXCLUDE_FROM_ALL)
add_library(ZLIB::ZLIB ALIAS zlibstatic)
target_include_directories(zlibstatic PUBLIC "${ZLIB_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/zlib")
include("${ENGINE_DIR}/cmake/ncine_sources.cmake")
function(ncine_assign_source_group)
endfunction()
add_subdirectory("${NCINE_SOURCE_DIR}/Utilities/AssetPacker" "${CMAKE_BINARY_DIR}/packer")
if(MINGW)
    # Ubuntu 20.04's MinGW defaults to Windows XP declarations; the shared
    # base layer uses Windows 8 APIs, including process token pseudo-handles.
    target_compile_definitions(AssetPacker PRIVATE _WIN32_WINNT=0x0602 WINVER=0x0602)
endif()
